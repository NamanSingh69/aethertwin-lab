cmake_minimum_required(VERSION 3.18)
project(lab_sensors_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_CMAKE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${PYBIND_CMAKE}")
find_package(pybind11 CONFIG REQUIRED)
option(WITH_UNITREE "Build Unitree L2 adapter" ON)
option(WITH_SLAMTEC "Build Slamtec S2 adapter" OFF)
set(UNITREE_SDK "" CACHE PATH "Path to unilidar_sdk2/unitree_lidar_sdk")
set(SLAMTEC_SDK "" CACHE PATH "Path to rplidar_sdk/sdk")
set(SLAMTEC_LIBRARY "" CACHE FILEPATH "Built official Slamtec SDK static library")
pybind11_add_module(lab_sensors_native bindings.cpp)
if(WITH_UNITREE)
  if(NOT UNIX)
    message(FATAL_ERROR "Unitree prebuilt SDK integration is supported on Ubuntu; use replay on Windows")
  endif()
  find_library(UNITREE_LIBRARY unilidar_sdk2 PATHS "${UNITREE_SDK}/lib/${CMAKE_SYSTEM_PROCESSOR}" NO_DEFAULT_PATH REQUIRED)
  target_include_directories(lab_sensors_native PRIVATE "${UNITREE_SDK}/include")
  target_link_libraries(lab_sensors_native PRIVATE "${UNITREE_LIBRARY}" pthread)
  target_compile_definitions(lab_sensors_native PRIVATE WITH_UNITREE)
endif()
if(WITH_SLAMTEC)
  if(NOT EXISTS "${SLAMTEC_LIBRARY}")
    message(FATAL_ERROR "Build the official Slamtec SDK and specify SLAMTEC_LIBRARY")
  endif()
  target_include_directories(lab_sensors_native PRIVATE "${SLAMTEC_SDK}/include")
  target_link_libraries(lab_sensors_native PRIVATE "${SLAMTEC_LIBRARY}")
  target_compile_definitions(lab_sensors_native PRIVATE WITH_SLAMTEC)
  if(UNIX)
    target_link_libraries(lab_sensors_native PRIVATE pthread)
  else()
    target_link_libraries(lab_sensors_native PRIVATE ws2_32 winmm)
  endif()
endif()
install(TARGETS lab_sensors_native LIBRARY DESTINATION . RUNTIME DESTINATION .)
